# a simple script that just makes it slightly faster to install blepfx's plugins.

## install and pick the plugins you want
curl https://raw.githubusercontent.com/echoe/blepfx_plugin_multiinstaller/refs/heads/main/install_blepfx.sh -o install_blepfx.sh; sh install_blepfx.sh

## install everything immediately
curl https://raw.githubusercontent.com/echoe/blepfx_plugin_multiinstaller/refs/heads/main/install_blepfx.sh -o install_blepfx.sh; sh install_blepfx.sh 12345 f

## you may want to delete the script after
rm install_blepfx.sh
